#include <stdio.h>

int main()
{
    int a[100], n, i, largest, second;

    printf("Enter number of elements: ");
    scanf("%d", &n);

    if(n < 2 || n > 100)
    {
        printf("Invalid array size");
        return 0;
    }

    printf("Enter array elements:\n");
    for(i = 0; i < n; i++)
    {
        scanf("%d", &a[i]);
    }

    largest = a[0];

    for(i = 1; i < n; i++)
    {
        if(a[i] > largest)
        {
            largest = a[i];
        }
    }

    second = a[0];

    for(i = 0; i < n; i++)
    {
        if(a[i] != largest)
        {
            second = a[i];
            break;
        }
    }

    if(i == n)
    {
        printf("No second largest element");
        return 0;
    }

    for(i = 0; i < n; i++)
    {
        if(a[i] > second && a[i] < largest)
        {
            second = a[i];
        }
    }

    printf("Second largest element = %d", second);

    return 0;
}
