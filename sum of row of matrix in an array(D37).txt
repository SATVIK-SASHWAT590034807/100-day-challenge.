#include <stdio.h>

int main()
{
    int a[10][10], sum[10], i, j, r, c;

    printf("Enter number of rows: ");
    scanf("%d", &r);

    printf("Enter number of columns: ");
    scanf("%d", &c);

    if(r < 1 || r > 10 || c < 1 || c > 10)
    {
        printf("Invalid matrix size");
        return 0;
    }

    printf("Enter matrix elements:\n");

    for(i = 0; i < r; i++)
    {
        for(j = 0; j < c; j++)
        {
            scanf("%d", &a[i][j]);
        }
    }

    for(i = 0; i < r; i++)
    {
        sum[i] = 0;

        for(j = 0; j < c; j++)
        {
            sum[i] = sum[i] + a[i][j];
        }
    }

    printf("Sum of each row:\n");

    for(i = 0; i < r; i++)
    {
        printf("Row %d = %d\n", i + 1, sum[i]);
    }

    return 0;
}
