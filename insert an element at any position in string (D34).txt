#include <stdio.h>

int main()
{
    int a[100], n, i, pos, num;

    printf("Enter number of elements: ");
    scanf("%d", &n);

    printf("Enter array elements:\n");
    for(i = 0; i < n; i++)
    {
        scanf("%d", &a[i]);
    }

    printf("Enter position: ");
    scanf("%d", &pos);

    printf("Enter element to insert: ");
    scanf("%d", &num);

    if(pos < 1 || pos > n + 1 || n >= 100)
    {
        printf("Invalid position or array is full");
        return 0;
    }

    for(i = n - 1; i >= pos - 1; i--)
    {
        a[i + 1] = a[i];
    }

    a[pos - 1] = num;
    n++;

    printf("New array:\n");
    for(i = 0; i < n; i++)
    {
        printf("%d ", a[i]);
    }

    return 0;
}
